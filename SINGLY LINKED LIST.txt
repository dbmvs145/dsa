SINGLY LINKED LIST

#include <stdio.h>
#include <stdlib.h>

struct Node
{
    int data;
    struct Node* next;
};

struct Node* head = NULL;

struct Node* createNode(int val)
{
    struct Node* newNode = (struct Node*)malloc(sizeof(struct Node));
    newNode->data = val;
    newNode->next = NULL;
    return newNode;
}

void createList()
{
    int n, val, i;
    struct Node* newNode;
    struct Node* temp;

    if (head != NULL)
    {
        head = NULL;
    }
    printf("Enter number of nodes: ");
    scanf("%d", &n);
    for (i = 0; i < n; i++)
    {
        printf("Enter data %d: ", i+1);
        scanf("%d", &val);
        newNode = createNode(val);
        if (head == NULL)
        {
            head = newNode;
            temp = head;
        }
        else
        {
            temp->next = newNode;
            temp = newNode;
        }
    }
    printf("List created\n");
}

void printList()
{
    struct Node* temp;
    if (head == NULL)
    {
        printf("List is EMPTY\n");
        return;
    }
    printf("List: ");
    temp = head;
    while (temp != NULL)
    {
        printf("%d -> ", temp->data);
        temp = temp->next;
    }
    printf("NULL\n");
}

void insertFirst(int val)
{
    struct Node* newNode = createNode(val);
    newNode->next = head;
    head = newNode;
    printf("%d inserted at first\n", val);
}

void insertLast(int val)
{
    struct Node* newNode = createNode(val);
    struct Node* temp;
    if (head == NULL)
    {
        head = newNode;
    }
    else
    {
        temp = head;
        while (temp->next != NULL)
        {
            temp = temp->next;
        }
        temp->next = newNode;
    }
    printf("%d inserted at last\n", val);
}

void insertAtPosition(int pos, int val)
{
    int i;
    struct Node* newNode;
    struct Node* temp;
    if (pos <= 0)
    {
        printf("Invalid position\n");
        return;
    }
    if (pos == 1)
    {
        insertFirst(val);
        return;
    }
    newNode = createNode(val);
    temp = head;
    for (i = 1; i < pos-1 && temp != NULL; i++)
    {
        temp = temp->next;
    }
    if (temp == NULL)
    {
        printf("Invalid position\n");
        free(newNode);
        return;
    }
    newNode->next = temp->next;
    temp->next = newNode;
    printf("%d inserted at pos %d\n", val, pos);
}

void insertAfterData(int key, int val)
{
    struct Node* temp = head;
    struct Node* newNode;
    while (temp != NULL && temp->data != key)
    {
        temp = temp->next;
    }
    if (temp == NULL)
    {
        printf("Data %d not found\n", key);
        return;
    }
    newNode = createNode(val);
    newNode->next = temp->next;
    temp->next = newNode;
    printf("%d inserted after %d\n", val, key);
}

void deleteFirst()
{
    struct Node* temp;
    if (head == NULL)
    {
        printf("List EMPTY\n");
        return;
    }
    temp = head;
    head = head->next;
    printf("%d deleted from first\n", temp->data);

}

void deleteLast()
{
    struct Node* temp;
    struct Node* prev;
    if (head == NULL)
    {
        printf("List EMPTY\n");
        return;
    }
    if (head->next == NULL)
    {
        printf("%d deleted from last\n", head->data);

        head = NULL;
        return;

    temp = head;
    while (temp->next != NULL)
    {
        prev = temp;
        temp = temp->next;
    }
    prev->next = NULL;
    printf("%d deleted from last\n", temp->data);
    free(temp);

void deleteAtPosition(int pos)
{
    int i;
    struct Node* temp;
    struct Node* delNode;

    {
        printf("List EMPTY\n");

    {

    printf("%d deleted from pos %d\n", delNode->data, pos);

}

{

    else
    {
        prev->next = temp->next;

}

{

    struct Node* i;
    struct Node* j;

    {
        printf("List EMPTY\n");

    }
    for (i = head; i != NULL; i = i->next)
    {
        for (j = i->next; j != NULL; j = j->next)
        {
            if (i->data > j->data)
            {
                tempData = i->data;
                i->data = j->data;
                j->data = tempData;
            }
        }
    }
    printf("List sorted\n");
}

void reverseList()
{
    struct Node* prev = NULL;
    struct Node* curr = head;
    struct Node* next = NULL;
    if (head == NULL)
    {
        printf("List EMPTY\n");
        return;
    }
    while (curr != NULL)
    {
        next = curr->next;
        curr->next = prev;
        prev = curr;
        curr = next;
    }
    head = prev;
    printf("List reversed\n");
}

#include <stdlib.h>

struct Node
{
    int data;
    struct Node* prev;
    struct Node* next;
};

struct Node* head = NULL;

struct Node* createNode(int val)
{
    struct Node* newNode = (struct Node*)malloc(sizeof(struct Node));
    newNode->data = val;
    newNode->prev = NULL;
    newNode->next = NULL;
    return newNode;
}

void createList()
{
    int n, val, i;
    struct Node* newNode;
    struct Node* temp;

    head = NULL;
    printf("Enter number of nodes: ");
    scanf("%d", &n);
    for (i = 0; i < n; i++)
    {
        printf("Enter data %d: ", i+1);
        scanf("%d", &val);
        newNode = createNode(val);
        if (head == NULL)
        {
            head = newNode;
            temp = head;

        {
            temp->next = newNode;

}

HASHING

#include <stdio.h>
#include <stdlib.h>
#define SIZE 10

int hashTable[SIZE];
int flag[SIZE]; // 0=empty, 1=occupied, 2=deleted

void initTable()
{
    int i;
    for (i = 0; i < SIZE; i++)
    {
        flag[i] = 0;
        hashTable[i] = 0;
    }
    printf("Hash Table initialized (Size %d)\n", SIZE);
}

int hashFunction(int key)
{
    return key % SIZE;
}

void insert(int key)
{
    int i, index, h;
    h = hashFunction(key);

    for (i = 0; i < SIZE; i++)
    {
        index = (h + i) % SIZE;
        if (flag[index] == 0 || flag[index] == 2)
        {
            hashTable[index] = key;
            flag[index] = 1;
            printf("%d inserted at index %d\n", key, index);
            return;
        }
    }
    printf("Hash Table FULL, cannot insert %d\n", key);
}

void search(int key)
{
    int i, index, h;
    h = hashFunction(key);

    for (i = 0; i < SIZE; i++)
    {
        index = (h + i) % SIZE;
        if (flag[index] == 0)
        {
            printf("%d NOT FOUND\n", key);
            return;
        }
        if (flag[index] == 1 && hashTable[index] == key)
        {
            printf("%d FOUND at index %d\n", key, index);
            return;
        }
    }
    printf("%d NOT FOUND\n", key);
}

void deleteKey(int key)
{
    int i, index, h;
    h = hashFunction(key);

    for (i = 0; i < SIZE; i++)
    {
        index = (h + i) % SIZE;
        if (flag[index] == 0)
        {
            printf("%d NOT FOUND\n", key);
            return;
        }
        if (flag[index] == 1 && hashTable[index] == key)
        {
            flag[index] = 2;
            printf("%d deleted from index %d\n", key, index);
            return;
        }
    }
    printf("%d NOT FOUND\n", key);
}

void display()
{
    int i;
    printf("\n--- Hash Table ---\n");
    printf("Index  Key  Status\n");
    for (i = 0; i < SIZE; i++)
    {
        if (flag[i] == 1)
        {
            printf("%5d %5d  Occupied\n", i, hashTable[i]);
        }
        else if (flag[i] == 2)
        {
            printf("%5d  ----  Deleted\n", i);
        }
        else
        {
            printf("%5d  ----  Empty\n", i);
        }
    }
}

{

    {

    }

void heapifyMax(int size, int i)
{
    int largest = i;
    int left = 2 * i;
    int right = 2 * i + 1;
    int temp;

    if (left <= size && arr[left] > arr[largest])
    {
        largest = left;
    }
    if (right <= size && arr[right] > arr[largest])
    {
        largest = right;
    }
    if (largest != i)
    {
        temp = arr[i];
        arr[i] = arr[largest];
        arr[largest] = temp;
        heapifyMax(size, largest);
    }
}

void heapifyMin(int size, int i)
{
    int smallest = i;
    int left = 2 * i;
    int right = 2 * i + 1;
    int temp;

    if (left <= size && arr[left] < arr[smallest])
    {
        smallest = left;
    }
    if (right <= size && arr[right] < arr[smallest])
    {
        smallest = right;
    }
    if (smallest != i)
    {
        temp = arr[i];
        arr[i] = arr[smallest];
        arr[smallest] = temp;
        heapifyMin(size, smallest);
    }
}

void heapSortAscending()
{
    int i, temp;
    for (i = n/2; i >= 1; i--)

        arr[1] = arr[i];

        heapifyMax(i-1, 1);

    printf("Sorted Ascending using Heap Sort\n");
}

BINARY SEARCH

#include <stdio.h>
#include <stdlib.h>
#define MAX 100

int arr[MAX];
int n;

void sortAscending()
{
    int i, j, temp;
    for (i = 0; i < n-1; i++)
    {
        for (j = 0; j < n-i-1; j++)
        {
            if (arr[j] > arr[j+1])
            {
                temp = arr[j];
                arr[j] = arr[j+1];
                arr[j+1] = temp;
            }
        }
    }
}

int binarySearch(int key)
{
    int low = 0;
    int high = n - 1;
    int mid;

    while (low <= high)
    {
        mid = (low + high) / 2;

        if (arr[mid] == key)
        {
            return mid;
        }
        else if (arr[mid] < key)
        {
            low = mid + 1;
        }
        else
        {
            high = mid - 1;
        }
    }
    return -1;
}

void display()
{
    int i;
    printf("Array: ");
    for (i = 0; i < n; i++)
    {
        printf("%d ", arr[i]);
    }
    printf("\n");
}

        printf("3.Search Element\n");

        switch (choice)

                display();
                break;

                    printf("%d found at index %d\n", key, result);

                }
                break;
            case 4:

        scanf("%d", &arr[i]);
    }
    arr[0] = 0; // reserve for sentinel
}

// 1. MODIFIED BUBBLE SORT
void modifiedBubbleSort()
{
    int i, j, temp;
    int swapped;
    for (i = 1; i <= n-1; i++)
    {
        swapped = 0;
        for (j = 1; j <= n-i; j++)
        {
            if (arr[j] > arr[j+1])
            {
                temp = arr[j];
                arr[j] = arr[j+1];
                arr[j+1] = temp;
                swapped = 1;
            }
        }
        if (swapped == 0)
        {
            printf("Early exit at pass %d - already sorted\n", i);
            break;
        }
    }
    printf("Sorted using Modified Bubble Sort\n");
}

// 2. INSERTION SORT USING SENTINEL
void insertionSortSentinel()
{
    int i, j;
    for (i = 2; i <= n; i++)
    {
        arr[0] = arr[i]; // sentinel stores current
        j = i - 1;
        while (arr[j] > arr[0])
        {
            arr[j+1] = arr[j];
            j--;
        }
        arr[j+1] = arr[0];
    }
    printf("Sorted using Insertion Sort with Sentinel\n");
}

// 3. SHELL SORT
void shellSort()
{
    int gap, i, j, temp;
    for (gap = n/2; gap >= 1; gap = gap/2)
    {
        for (i = gap+1; i <= n; i++)
        {
            temp = arr[i];
            j = i - gap;
            while (j >= 1 && arr[j] > temp)
            {
                arr[j+gap] = arr[j];
                j = j - gap;
            }
            arr[j+gap] = temp;
        }
        printf("After gap %d: ", gap);
        display();
    }
    printf("Sorted using Shell Sort\n");
}

IDENTITY DIAGONAL

#include <stdio.h>
#include <stdlib.h>
#define MAX 10

int mat[MAX][MAX];
int r, c;

void inputMatrix()
{
    int i, j;
    printf("Enter rows and cols: ");
    scanf("%d %d", &r, &c);
    printf("Enter %d x %d matrix:\n", r, c);
    for (i = 0; i < r; i++)
    {
        for (j = 0; j < c; j++)
        {
            scanf("%d", &mat[i][j]);
        }
    }
}

void displayMatrix()
{
    int i, j;
    if (r == 0)
    {
        printf("Matrix EMPTY\n");
        return;
    }
    printf("Matrix:\n");
    for (i = 0; i < r; i++)
    {
        for (j = 0; j < c; j++)
        {
            printf("%d ", mat[i][j]);
        }
        printf("\n");
    }
}

int checkDiagonal()
{
    int i, j;
    if (r != c)
    {
        return 0;
    }
    for (i = 0; i < r; i++)
    {
        for (j = 0; j < c; j++)
        {
            if (i != j && mat[i][j] != 0)
            {
                return 0;
            }
        }
    }
    return 1;
}

int checkIdentity()
{
    int i, j;
    if (r != c)
    {
        return 0;
    }
    for (i = 0; i < r; i++)
    {
        for (j = 0; j < c; j++)
        {
            if (i == j && mat[i][j] != 1)
            {
                return 0;
            }
            if (i != j && mat[i][j] != 0)
            {
                return 0;
            }
        }
    }
    return 1;
}

{
    int choice;

    inputMatrix();

    while (1)
    {
        printf("\n--- MATRIX CHECK MENU ---\n");

        printf("3.Check Diagonal\n");
        printf("4.Check Identity\n");
        printf("5.Check Both\n");
        printf("6.Exit\n");
        printf("Enter choice: ");
        scanf("%d", &choice);

        switch (choice)
        {
            case 1:
                inputMatrix();
                break;
            case 2:
                displayMatrix();
                break;
            case 3:
                if (checkDiagonal())
                {
                    printf("It is DIAGONAL Matrix\n");
                }
                else
                {
                    printf("It is NOT Diagonal Matrix\n");
                }
                break;
            case 4:
                if (checkIdentity())
                {
                    printf("It is IDENTITY Matrix\n");
                }
                else
                {
                    printf("It is NOT Identity Matrix\n");
                }
                break;
            case 5:
                if (checkIdentity())
                {
                    printf("It is IDENTITY and also DIAGONAL\n");
                }
                else if (checkDiagonal())
                {
                    printf("It is DIAGONAL but NOT Identity\n");
                }
                else
                {
                    printf("It is NEITHER Diagonal nor Identity\n");
                }
                break;
            case 6:
                exit(0);
                break;
            default:
                printf("Invalid Choice\n");
                break;
        }
    }
